cmake_minimum_required(VERSION 3.20)
project(Knock VERSION 0.6.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()

add_library(knock_core src/store.cpp)
target_include_directories(knock_core PUBLIC include third_party)
add_library(knock_settings src/window_settings.cpp)
target_include_directories(knock_settings PUBLIC include PRIVATE third_party)
if(MSVC)
    target_compile_options(knock_core PRIVATE /W4 /permissive- /utf-8)
    target_compile_options(knock_settings PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(knock_core PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(knock_settings PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Keep the in-terminal instructions and the distributed Markdown identical.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_SOURCE_DIR}/docs/AGENT_INSTRUCTIONS.md")
file(READ "${CMAKE_SOURCE_DIR}/docs/AGENT_INSTRUCTIONS.md" KNOCK_AGENT_INFO)
string(FIND "${KNOCK_AGENT_INFO}" ")knock_info\"" guide_delimiter)
if(NOT guide_delimiter EQUAL -1)
    message(FATAL_ERROR "Agent instructions contain the C++ raw-string delimiter")
endif()
configure_file(scripts/agent_info.hpp.in generated/knock/agent_info.hpp @ONLY)

include(CTest)
if(BUILD_TESTING)
    add_executable(cli_info_tests tests/cli_info_tests.cpp src/cli.cpp)
    target_include_directories(cli_info_tests PRIVATE include third_party "${CMAKE_CURRENT_BINARY_DIR}/generated")
    if(MSVC)
        target_compile_options(cli_info_tests PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(cli_info_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME cli_info COMMAND cli_info_tests)
    set_tests_properties(cli_info PROPERTIES TIMEOUT 10)
    add_executable(store_tests tests/store_tests.cpp)
    target_link_libraries(store_tests PRIVATE knock_core)
    add_test(NAME store COMMAND store_tests)
    add_executable(window_settings_tests tests/window_settings_tests.cpp)
    target_include_directories(window_settings_tests PRIVATE third_party)
    target_link_libraries(window_settings_tests PRIVATE knock_settings)
    if(MSVC)
        target_compile_options(window_settings_tests PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(window_settings_tests PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME window_settings COMMAND window_settings_tests)
    set_tests_properties(window_settings PROPERTIES TIMEOUT 30)
endif()

if(WIN32)
    enable_language(RC)
    add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
    add_library(knock_platform src/platform_win.cpp src/ipc_win.cpp)
    target_link_libraries(knock_platform PUBLIC knock_core advapi32 shell32 ole32 user32)
    if(BUILD_TESTING)
        add_executable(platform_tests tests/platform_tests.cpp)
        target_link_libraries(platform_tests PRIVATE knock_platform)
        add_test(NAME platform COMMAND platform_tests)
        set_tests_properties(platform PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
    endif()
    add_executable(KnockTray WIN32 src/tray_main.cpp src/ui.cpp src/sprite_win.cpp
        src/hotkey_win.cpp src/note_images_win.cpp assets/app.rc)
    target_link_libraries(KnockTray PRIVATE knock_platform knock_settings gdiplus comctl32 dwmapi user32 gdi32)
    add_executable(knock src/cli_main.cpp src/cli.cpp src/runner_win.cpp assets/cli.rc)
    target_link_libraries(knock PRIVATE knock_platform)
    target_include_directories(knock PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
    if(BUILD_TESTING)
        add_executable(sprite_tests tests/sprite_tests.cpp src/sprite_win.cpp)
        target_include_directories(sprite_tests PRIVATE include)
        target_link_libraries(sprite_tests PRIVATE gdiplus gdi32 user32)
        add_dependencies(sprite_tests KnockTray)
        add_test(NAME sprites COMMAND sprite_tests)
        set_tests_properties(sprites PROPERTIES TIMEOUT 30)
        add_executable(note_images_tests tests/note_images_tests.cpp src/note_images_win.cpp)
        target_include_directories(note_images_tests PRIVATE include)
        target_link_libraries(note_images_tests PRIVATE gdiplus ole32 gdi32 user32)
        add_test(NAME note_images COMMAND note_images_tests)
        set_tests_properties(note_images PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
        add_executable(hotkey_tests tests/hotkey_tests.cpp src/hotkey_win.cpp)
        target_include_directories(hotkey_tests PRIVATE include third_party)
        target_link_libraries(hotkey_tests PRIVATE user32 gdi32)
        add_test(NAME hotkey COMMAND hotkey_tests)
        set_tests_properties(hotkey PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
        add_executable(runner_fixture tests/runner_fixture.cpp)
        target_link_libraries(runner_fixture PRIVATE knock_core)
        add_executable(inbox_ui_tests tests/inbox_ui_tests.cpp)
        target_link_libraries(inbox_ui_tests PRIVATE knock_platform gdi32)
        if(MSVC)
            target_compile_options(inbox_ui_tests PRIVATE /W4 /permissive- /utf-8)
        else()
            target_compile_options(inbox_ui_tests PRIVATE -Wall -Wextra -Wpedantic)
        endif()
        add_dependencies(inbox_ui_tests KnockTray)
        add_test(NAME inbox_ui COMMAND inbox_ui_tests)
        set_tests_properties(inbox_ui PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
        add_executable(notes_ui_tests tests/notes_ui_tests.cpp)
        target_link_libraries(notes_ui_tests PRIVATE knock_platform gdi32)
        if(MSVC)
            target_compile_options(notes_ui_tests PRIVATE /W4 /permissive- /utf-8)
        else()
            target_compile_options(notes_ui_tests PRIVATE -Wall -Wextra -Wpedantic)
        endif()
        add_dependencies(notes_ui_tests KnockTray)
        add_test(NAME notes_ui COMMAND notes_ui_tests)
        set_tests_properties(notes_ui PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
        add_executable(ui_tests tests/ui_tests.cpp)
        target_link_libraries(ui_tests PRIVATE knock_platform)
        if(MSVC)
            target_compile_options(ui_tests PRIVATE /W4 /permissive- /utf-8)
        else()
            target_compile_options(ui_tests PRIVATE -Wall -Wextra -Wpedantic)
        endif()
        add_dependencies(ui_tests KnockTray knock)
        add_test(NAME ui COMMAND ui_tests)
        set_tests_properties(ui PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77 RUN_SERIAL TRUE)
    endif()
    foreach(target knock_platform KnockTray knock)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endforeach()
    if(MINGW)
        foreach(target KnockTray knock cli_info_tests store_tests window_settings_tests platform_tests inbox_ui_tests notes_ui_tests ui_tests sprite_tests runner_fixture note_images_tests hotkey_tests)
            if(TARGET ${target})
                target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
            endif()
        endforeach()
        target_link_options(KnockTray PRIVATE -municode)
        target_link_options(knock PRIVATE -municode)
        if(TARGET runner_fixture)
            target_link_options(runner_fixture PRIVATE -municode)
        endif()
    endif()
    foreach(asset request.png complete.png note.png request.gif complete.gif note.gif)
        add_custom_command(TARGET KnockTray POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:KnockTray>/assets" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/assets/${asset}" "$<TARGET_FILE_DIR:KnockTray>/assets/${asset}")
    endforeach()
    install(TARGETS KnockTray knock RUNTIME DESTINATION .)
    install(FILES assets/request.png assets/complete.png assets/note.png
        assets/request.gif assets/complete.gif assets/note.gif DESTINATION assets)
    install(FILES README.md docs/CLI.md docs/AGENT_INSTRUCTIONS.md THIRD_PARTY_NOTICES.md DESTINATION .)
    install(FILES HANDOFF.md DESTINATION .)
    install(FILES docs/VALIDATION.md docs/PROTOCOL.md docs/FEATURES.md
        docs/CLI.md docs/AGENT_INSTRUCTIONS.md DESTINATION docs)
    install(FILES docs/screenshots/inbox.png docs/screenshots/note-conversation.png
        docs/screenshots/note-find.png docs/screenshots/settings.png
        docs/screenshots/saved-replies.png docs/screenshots/note-entry-edit.png
        docs/screenshots/note-trash.png docs/screenshots/preferences.png
        DESTINATION docs/screenshots)
    install(FILES third_party/nlohmann-LICENSE.txt DESTINATION licenses)
endif()
